employee_db=# SELECT
    p.id AS project_id,
    p.name AS project_name,
    t.id AS task_id,
    t.title AS task_title
FROM projects p
JOIN tasks t
    ON t.project_id = p.id;
 project_id | project_name | task_id | task_title 
------------+--------------+---------+------------
          1 | My Project   |       1 | First Task
(1 row)

employee_db=# SELECT
    e.id AS employee_id,
    e.name AS employee_name,
    p.id AS project_id,
    p.name AS project_name,
    m.role
FROM employees e
JOIN memberships m
    ON m.employee_id = e.id
JOIN projects p
    ON p.id = m.project_id;
 employee_id | employee_name | project_id | project_name | role 
-------------+---------------+------------+--------------+------
(0 rows)

employee_db=# SELECT
    t.id AS task_id,
    t.title,
    tg.name AS tag
FROM tasks t
JOIN task_tags tt
    ON tt.task_id = t.id
JOIN tags tg
    ON tg.id = tt.tag_id;
 task_id | title | tag 
---------+-------+-----
(0 rows)
